ubuntu@ip-172-33-86-172:~$ set -u

REGION="eu-west-2"
ACL_NAME="CreatedByALB-magento-prod-ALB"
LOG_GROUP="aws-waf-logs-magento-prod"

echo "=== WAF Web ACL ==="
aws wafv2 list-web-acls \
  --scope REGIONAL \
  --region "$REGION" \
  --query "WebACLs[?Name=='$ACL_NAME'].[Name,Id,ARN]" \
  --output table

ACL_ID=$(aws wafv2 list-web-acls \
  --scope REGIONAL \
  --region "$REGION" \
  --query "WebACLs[?Name=='$ACL_NAME'].Id | [0]" \
  --output text)

ACL_ARN=$(aws wafv2 list-web-acls \
  --scope REGIONAL \
  --region "$REGION" \
  --query "WebACLs[?Name=='$ACL_NAME'].ARN | [0]" \
  --output text)

echo
echo "=== WAF configuration ==="
aws wafv2 get-web-acl \
  --name "$ACL_NAME" \
  --scope REGIONAL \
  --id "$ACL_ID" \
  --region "$REGION" \
  --query "WebACL.[Name,Id,ARN,DefaultAction,Capacity]" \
  --output json

echo
echo "=== ALB association ==="
aws wafv2 list-resources-for-web-acl \
  --web-acl-arn "$ACL_ARN" \
  --output tablereams[].{name:logStreamName,lastEvent:lastEventTimestamp}" \storedBytes:storedBytes}" \
