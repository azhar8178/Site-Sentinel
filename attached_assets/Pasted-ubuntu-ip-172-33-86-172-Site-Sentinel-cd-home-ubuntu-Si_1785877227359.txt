SELECT
  count(*) AS oauth_rows,
  count(*) FILTER (WHERE client_id <> '') AS rows_with_client_id,
  count(*) FILTER (WHERE client_secret <> '') AS rows_with_client_secret,
  max(updated_at) AS last_oauth_update
FROM google_oauth_config;

SELECT
  count(*) AS token_rows,
  count(*) FILTER (WHERE refresh_token IS NOT NULL AND refresh_token <> '') AS refresh_token_rows,
  max(created_at) AS last_token_created,
  max(ga_property_id) AS selected_property
FROM google_analytics_tokens;

SELECT
  count(*) AS monitored_servers,
  (SELECT count(*) FROM server_metrics) AS metric_rows
FROM servers;
"
